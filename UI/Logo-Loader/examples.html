// React component (single SVG, no size variants)
import LoaderSvg from './loader.svg?react';

const PX = { xs: 12, sm: 16, md: 24, lg: 40, xl: 64 };

export function Loader({ size = 'md', className }) {
  const px = PX[size];
  return (
    <LoaderSvg
      width={px} height={px}
      className={className}
      aria-label="Loading"
    />
  );
}

// Animation (cloche bob, steam, plated food) is baked
// into loader.svg. Cloche inherits text colour;
// food stays brand amber.
// usage — same loader, sized per context

<span className="chip">
  <Loader size="xs" /> Saving…
</span>

<button disabled>
  <Loader size="md" /> Saving…
</button>

// dark background — flip the text colour,
// food stays amber
<div style={{ color: 'var(--bone)' }}>
  <Loader size="xl" />
</div>
